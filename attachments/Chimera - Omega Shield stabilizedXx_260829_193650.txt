def generate_majorana(N):

for i in range(N):

vec = vec / torch.norm(vec)
ops.append(vec)
return ops

chi = generate_majorana(N)

REAL SYK HAMILTONIAN

def syk_hamiltonian(chi, N):
H = torch.zeros(N, N, device=device)
for i in range(N):
for j in range(i+1, N):
J = torch.randn(1, device=device) / np.sqrt(N)
H += J * torch.outer(chi[i], chi[j])
return (H + H.T) / 2
