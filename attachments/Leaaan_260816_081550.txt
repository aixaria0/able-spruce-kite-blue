ih] rw [count_pos_cons] cases hd with | gauge => rfl | phase p => cases p <;>

import Mathlib.Logic.Basic
import Mathlib.Data.Nat.Basic
import Mathlib.Tactic

namespace GodelLobLogic

opaque Provable : Prop → Prop
notation "□" φ:max => Provable φ

axiom axiom_K (φ ψ : Prop) : □ (φ → ψ) → (□ φ → □ ψ)
axiom axiom_4 (φ : Prop) : □ φ → □ (□ φ)
axiom axiom_Lob (φ : Prop) : □ (□ φ → φ) → □ φ

def IsConsistent : Prop := ¬ □ False

theorem Godel_Second_Incompleteness :
□ IsConsistent → ¬ IsConsistent := by
intro h_box_consis
have h_lob := axiom_Lob False
have h_box_false : □ False := h_lob h_box_consis
intro h_consis
exact h_consis h_box_false

/-- تعریف فروتنی (Soundness / Humility) -/
def IsHumble (φ : Prop) : Prop := □ φ → φ

/-- قضیه افق درستی:
سازگاری سیستم فروتن است اگر و تنها اگر اثبات‌ناپذیر باشد. -/
theorem Consistency_Humble_iff_Unprovable :
IsHumble IsConsistent ↔ ¬ □ IsConsistent := by
constructor
· -- جهت → : اگر سازگاری فروتن باشد، آنگاه اثبات‌ناپذیر است
intro h_humble
intro h_box
-- از فروتنی: □ IsConsistent → IsConsistent
have h_consis : IsConsistent := h_humble h_box
-- از قضیه G2: □ IsConsistent → ¬ IsConsistent
have h_not_consis : ¬ IsConsistent := Godel_Second_Incompleteness h_box
-- تناقض
exact h_not_consis h_consis
· -- جهت ← : اگر سازگاری اثبات‌ناپذیر باشد، آنگاه فروتن است
intro h_unprovable
intro h_box
-- از فرض ¬ □ IsConsistent و داشتن □ IsConsistent به تناقض می‌رسیم
have h_false : False := h_unprovable h_box
-- از False می‌توان هر چیزی نتیجه گرفت (از جمله IsConsistent)
exact False.elim h_false

end GodelLobLogic
